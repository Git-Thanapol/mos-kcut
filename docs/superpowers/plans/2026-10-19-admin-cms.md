# Admin CMS (Stage 2) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let the owner edit the landing-page content from a Thai `/admin` page; saving rebuilds the static site and publishes it atomically.

**Architecture:** `site/data/content.json` is read at build time by `getContent()` (falls back to `defaults.ts`). A small Fastify service (`site/server/`) authenticates the owner, validates and stores content, compresses uploads (sharp / ffmpeg), and runs `astro build` into a fresh `releases/<ts>` directory, then swaps the `dist` symlink. `public/admin.html` + `admin.js` is the plain-JS form.

**Tech Stack:** Node >= 22.18 (native TS stripping, `node:test`), Fastify 5, `@fastify/cookie`, `@fastify/multipart`, sharp, ffmpeg (system binary; `ffmpeg-static` for local/test), Astro 5.

**Spec:** `docs/superpowers/specs/2026-10-19-admin-cms-design.md`

**No git repo exists**, so tasks have no commit steps. All commands run from `site/` unless stated.

## Global Constraints

- Exactly 10 features, 6 clips, 3 reviews; counts are fixed (spec: Out of scope).
- All user-facing text (admin UI, API errors) is Thai.
- Secrets only from env: `ADMIN_PASSWORD`, `SESSION_SECRET`; only a placeholder `.env.example` is committed.
- Session cookie: `httpOnly`, `Secure` (unless `COOKIE_SECURE=false`), `SameSite=Strict`, 12 h. Mutating requests need header `X-Requested-With: kcut-admin`.
- Login rate limit: 5 failures per IP per 15 min, then 429.
- Upload limits: image 15 MB, video 300 MB; type decided by magic bytes, never by client MIME/filename; output names are generated UUIDs.
- Media paths in content: must match `/media/<file>` or be a path already present in `defaults.ts`; no `..`.
- Public page must stay visually identical to the prototype (no markup/class changes in components).
- Build failure never changes the live site; keep the last 3 releases.
- Server code is plain ESM `.mjs`, no build step; tests use `node --test`.

## Review Focus

1. Upload whose bytes do not match `?kind=` (PNG sent as `kind=video`, text file named `.mp4`) → 400, nothing left in `public/media`.
2. Media path traversal in a save (`/media/../../etc/passwd.png`) → 400.
3. Hand-edited / corrupt `data/content.json` (invalid JSON, wrong counts) → site still builds from defaults, no crash.
4. Thai text with newlines and emoji in `headline`/`body` is saved and read back byte-identical; length limit counts characters, not UTF-16 units.
5. Two saves while a build is running → exactly one extra build afterwards, status never stuck in `building`.
6. Wrong password 6 times → 6th gets 429, correct password during the block is still refused until the window ends.

```js
export const COUNTS = { features: 10, clips: 6, reviews: 3 };

const PATH_RE = /^\/[A-Za-z0-9_-][A-Za-z0-9_./-]*\.(webp|jpg|jpeg|png|mp4)$/;
const len = (s) => [...s].length;

const isObj = (v) => v !== null && typeof v === 'object' && !Array.isArray(v);

export function collectPaths(c) {
  const out = new Set();
  const add = (v) => { if (typeof v === 'string') out.add(v); };
  if (isObj(c?.hero)) { add(c.hero.image); add(c.hero.video); add(c.hero.poster); }
  for (const f of c?.features ?? []) add(f?.video);
  for (const k of c?.clips ?? []) add(k?.cover);
  for (const r of c?.reviews ?? []) add(r?.image);
  return out;
}

export function validateContent(c, known) {
  const errors = [];
  const bad = (field, message) => errors.push({ field, message });
  const text = (field, v, max) => {
    if (typeof v !== 'string' || len(v.trim()) < 1 || len(v) > max) bad(field, `ต้องไม่ว่างและไม่เกิน ${max} ตัวอักษร`);
  };
  const media = (field, v) => {
    const ok = typeof v === 'string' && PATH_RE.test(v) && !v.includes('..') && (v.startsWith('/media/') || known.has(v));
    if (!ok) bad(field, 'ไฟล์สื่อไม่ถูกต้อง');
  };

  if (!isObj(c)) return { ok: false, errors: [{ field: '', message: 'ข้อมูลไม่ถูกต้อง' }] };

  if (!isObj(c.hero)) bad('hero', 'ข้อมูลไม่ถูกต้อง');
  else { media('hero.image', c.hero.image); media('hero.video', c.hero.video); media('hero.poster', c.hero.poster); }

  for (const [key, n] of Object.entries(COUNTS)) {
    if (!Array.isArray(c[key]) || c[key].length !== n) bad(key, `จำนวนรายการต้องเป็น ${n}`);
  }
  if (errors.length) return { ok: false, errors };

  c.features.forEach((f, i) => {
    if (!isObj(f)) return bad(`features[${i}]`, 'ข้อมูลไม่ถูกต้อง');
    text(`features[${i}].title`, f.title, 120); text(`features[${i}].desc`, f.desc, 400); media(`features[${i}].video`, f.video);
  });
  c.clips.forEach((k, i) => {
    if (!isObj(k)) return bad(`clips[${i}]`, 'ข้อมูลไม่ถูกต้อง');
    text(`clips[${i}].views`, k.views, 20); text(`clips[${i}].title`, k.title, 120);
    text(`clips[${i}].subtitle`, k.subtitle, 120); media(`clips[${i}].cover`, k.cover);
  });
  c.reviews.forEach((r, i) => {
    if (!isObj(r)) return bad(`reviews[${i}]`, 'ข้อมูลไม่ถูกต้อง');
    if (!Number.isInteger(r.rating) || r.rating < 1 || r.rating > 5) bad(`reviews[${i}].rating`, 'คะแนนต้องเป็นเลข 1-5');
    text(`reviews[${i}].headline`, r.headline, 200); text(`reviews[${i}].name`, r.name, 120);
    text(`reviews[${i}].body`, r.body, 600); media(`reviews[${i}].image`, r.image);
  });
  return { ok: errors.length === 0, errors };
}
```

```ts
import fs from 'node:fs';
import path from 'node:path';
import { defaults, type Content } from './defaults';
import { validateContent, collectPaths } from '../../server/schema.mjs';

// Owner content (data/content.json, written by the admin service) overrides the defaults.
// Missing, unreadable or invalid file => defaults, so the build never breaks.
export async function getContent(): Promise<Content> {
  const file = path.resolve(process.cwd(), 'data/content.json');
  try {
    const parsed = JSON.parse(fs.readFileSync(file, 'utf8'));
    if (validateContent(parsed, collectPaths(defaults)).ok) return parsed as Content;
    console.warn('[content] data/content.json ไม่ผ่านการตรวจสอบ ใช้ค่าเริ่มต้นแทน');
  } catch (e) {
    if ((e as NodeJS.ErrnoException).code !== 'ENOENT') console.warn('[content] อ่าน data/content.json ไม่ได้ ใช้ค่าเริ่มต้นแทน');
  }
  return defaults;
}
```

```js
import { createHash, createHmac, timingSafeEqual } from 'node:crypto';

const sha = (s) => createHash('sha256').update(String(s)).digest();

export function checkPassword(input, expected) {
  // hash first so lengths are equal and the compare is constant time
  return timingSafeEqual(sha(input), sha(expected)) && String(expected).length > 0;
}

const mac = (secret, exp) => createHmac('sha256', secret).update(String(exp)).digest('hex');

export function signSession(secret, now = Date.now(), ttlMs = 12 * 3600 * 1000) {
  const exp = now + ttlMs;
  return `${exp}.${mac(secret, exp)}`;
}

export function verifySession(secret, token, now = Date.now()) {
  if (typeof token !== 'string') return false;
  const parts = token.split('.');
  if (parts.length !== 2) return false;
  const [exp, sig] = parts;
  if (!/^\d+$/.test(exp) || !/^[0-9a-f]{64}$/.test(sig)) return false;
  const good = Buffer.from(mac(secret, exp), 'hex');
  if (!timingSafeEqual(good, Buffer.from(sig, 'hex'))) return false;
  return Number(exp) > now;
}

export function createRateLimiter({ max = 5, windowMs = 15 * 60 * 1000 } = {}) {
  const hits = new Map(); // key -> { count, first }
  const live = (key, now) => {
    const h = hits.get(key);
    if (h && now - h.first >= windowMs) { hits.delete(key); return undefined; }
    return h;
  };
  return {
    isBlocked(key, now = Date.now()) { return (live(key, now)?.count ?? 0) >= max; },
    fail(key, now = Date.now()) {
      const h = live(key, now);
      if (h) h.count++; else hits.set(key, { count: 1, first: now });
    },
    reset(key) { hits.delete(key); },
  };
}
```

```js
import fsp from 'node:fs/promises';
import { spawn } from 'node:child_process';
import sharp from 'sharp';

export async function detectKind(file) {
  let fh;
  try {
    fh = await fsp.open(file, 'r');
    const b = Buffer.alloc(16);
    const { bytesRead } = await fh.read(b, 0, 16, 0);
    if (bytesRead < 12) return null;
    if (b[0] === 0xff && b[1] === 0xd8 && b[2] === 0xff) return 'image';                       // jpeg
    if (b.subarray(0, 8).equals(Buffer.from([0x89, 0x50, 0x4e, 0x47, 0x0d, 0x0a, 0x1a, 0x0a]))) return 'image'; // png
    if (b.toString('latin1', 0, 4) === 'RIFF' && b.toString('latin1', 8, 12) === 'WEBP') return 'image';
    if (b.toString('latin1', 4, 8) === 'ftyp') return 'video';                                  // mp4 / mov
    return null;
  } catch {
    return null;
  } finally {
    await fh?.close();
  }
}

export async function processImage(inFile, outFile) {
  await sharp(inFile).rotate().resize({ width: 1600, withoutEnlargement: true }).webp({ quality: 82 }).toFile(outFile);
}

export function processVideo(inFile, outFile) {
  const bin = process.env.FFMPEG_PATH || 'ffmpeg';
  const args = ['-y', '-i', inFile, '-vf', "scale=-2:'min(720,ih)'", '-c:v', 'libx264', '-crf', '28', '-preset', 'veryfast',
    '-pix_fmt', 'yuv420p', '-c:a', 'aac', '-b:a', '96k', '-movflags', '+faststart', outFile];
  return new Promise((resolve, reject) => {
    const p = spawn(bin, args, { stdio: ['ignore', 'ignore', 'pipe'] });
    let err = '';
    p.stderr.on('data', (d) => { err = (err + d).slice(-2000); });
    p.on('error', reject);
    p.on('close', (code) => (code === 0 ? resolve() : reject(new Error(`ffmpeg exited ${code}: ${err}`))));
  });
}
```

```js
export function createBuilder({ build, publish }) {
  let status = { state: 'idle', message: '', finishedAt: null };
  let current = null;
  let dirty = false;

  async function loop() {
    do {
      dirty = false;
      status = { state: 'building', message: '', finishedAt: null };
      try {
        const dir = await build();
        await publish(dir);
        status = { state: 'ok', message: '', finishedAt: new Date().toISOString() };
      } catch (e) {
        status = { state: 'failed', message: String(e?.message ?? e).slice(-500), finishedAt: new Date().toISOString() };
      }
    } while (dirty);
  }

      if (!current) {
        dirty = false;

      return current;
    },
    status: () => status,
  };
}
```

```js
import fsp from 'node:fs/promises';
import path from 'node:path';
import { spawn } from 'node:child_process';

export function makeBuild({ siteDir, releasesDir }) {
  return async () => {
    await fsp.mkdir(releasesDir, { recursive: true });
    const dir = path.join(releasesDir, new Date().toISOString().replace(/[:.]/g, '-'));
    const astro = path.join(siteDir, 'node_modules', 'astro', 'astro.js');
    await new Promise((resolve, reject) => {
      const p = spawn(process.execPath, [astro, 'build', '--outDir', dir], { cwd: siteDir, stdio: ['ignore', 'ignore', 'pipe'] });
      let err = '';
      p.stderr.on('data', (d) => { err = (err + d).slice(-2000); });
      p.on('error', reject);
      p.on('close', (code) => (code === 0 ? resolve() : reject(new Error(`astro build exited ${code}: ${err}`))));
    });
    return dir;
  };
}

export function makePublish({ siteDir, releasesDir, keep = 3 }) {
  const link = path.join(siteDir, 'dist');
  return async (dir) => {
    const st = await fsp.lstat(link).catch(() => null);
    if (st && !st.isSymbolicLink()) await fsp.rename(link, path.join(releasesDir, `initial-${Date.now()}`));

    const tmp = `${link}.tmp`;
    await fsp.rm(tmp, { force: true, recursive: true });
    await fsp.symlink(path.resolve(dir), tmp, 'junction');
    try {
      await fsp.rename(tmp, link);            // atomic on Linux
    } catch {
      await fsp.rm(link, { force: true, recursive: true });   // Windows dev fallback
      await fsp.rename(tmp, link);
    }

    const live = await fsp.realpath(link);
    const entries = await Promise.all((await fsp.readdir(releasesDir)).map(async (name) => {
      const full = path.join(releasesDir, name);
      return { full, mtime: (await fsp.stat(full)).mtimeMs };
    }));
    entries.sort((a, b) => b.mtime - a.mtime);
    for (const e of entries.slice(keep)) {
      if (path.resolve(e.full) !== path.resolve(live)) await fsp.rm(e.full, { recursive: true, force: true });
    }
  };
}
```

```js
import Fastify from 'fastify';
import cookie from '@fastify/cookie';
import multipart from '@fastify/multipart';
import { randomUUID } from 'node:crypto';
import fs from 'node:fs';
import fsp from 'node:fs/promises';
import path from 'node:path';
import { pipeline } from 'node:stream/promises';
import { checkPassword, signSession, verifySession, createRateLimiter } from './auth.mjs';
import { validateContent, collectPaths } from './schema.mjs';
import { detectKind, processImage, processVideo } from './media.mjs';

const LIMITS = { image: 15 * 1024 * 1024, video: 300 * 1024 * 1024 };
const COOKIE = 'kcut_session';

export function createApp({ env, builder, dataDir, mediaDir, defaults, processors = { image: processImage, video: processVideo } }) {
  const app = Fastify({ logger: false, trustProxy: true });
  const limiter = createRateLimiter();
  const known = collectPaths(defaults);
  const uploadsDir = path.join(dataDir, 'uploads');
  const contentFile = path.join(dataDir, 'content.json');

  app.register(cookie);
  app.register(multipart, { limits: { files: 1, fileSize: LIMITS.video } });

  app.addHook('preHandler', async (req, reply) => {
    const url = req.url.split('?')[0];
    if (!url.startsWith('/api/admin/')) return;
    if (req.method !== 'GET' && req.headers['x-requested-with'] !== 'kcut-admin') {
      return reply.code(403).send({ error: 'คำขอไม่ถูกต้อง' });
    }
    if (url === '/api/admin/login') return;
    if (!verifySession(env.SESSION_SECRET, req.cookies?.[COOKIE])) {
      return reply.code(401).send({ error: 'กรุณาเข้าสู่ระบบ' });
    }
  });

  const cookieOpts = { path: '/api/admin', httpOnly: true, sameSite: 'strict', secure: env.COOKIE_SECURE !== 'false' };

  app.post('/api/admin/login', async (req, reply) => {
    const key = req.ip;
    if (limiter.isBlocked(key)) return reply.code(429).send({ error: 'ลองผิดหลายครั้งเกินไป กรุณารอ 15 นาที' });
    if (!checkPassword(req.body?.password ?? '', env.ADMIN_PASSWORD)) {
      limiter.fail(key);
      return reply.code(401).send({ error: 'รหัสผ่านไม่ถูกต้อง' });
    }
    limiter.reset(key);
    reply.setCookie(COOKIE, signSession(env.SESSION_SECRET), { ...cookieOpts, maxAge: 12 * 3600 });
    return { ok: true };
  });

  app.post('/api/admin/logout', async (req, reply) => {
    reply.clearCookie(COOKIE, cookieOpts);
    return { ok: true };
  });

  async function readContent() {
    try {
      const parsed = JSON.parse(await fsp.readFile(contentFile, 'utf8'));
      if (validateContent(parsed, known).ok) return parsed;
    } catch { /* missing or corrupt -> defaults */ }
    return defaults;
  }

  app.get('/api/admin/content', async () => ({ content: await readContent(), defaults }));

  app.put('/api/admin/content', async (req, reply) => {
    const body = req.body;
    const v = validateContent(body, known);
    const errors = [...v.errors];
    if (v.ok) {
      for (const p of collectPaths(body)) {
        if (p.startsWith('/media/') && !fs.existsSync(path.join(mediaDir, path.basename(p)))) {
          errors.push({ field: p, message: 'ไม่พบไฟล์ที่อัปโหลด' });
        }
      }
    }
    if (errors.length) return reply.code(400).send({ error: 'ข้อมูลไม่ถูกต้อง', errors });
    await fsp.mkdir(dataDir, { recursive: true });
    const tmp = `${contentFile}.tmp`;
    await fsp.writeFile(tmp, JSON.stringify(body, null, 2), 'utf8');
    await fsp.rename(tmp, contentFile);
    builder.trigger();
    return reply.code(202).send({ ok: true });
  });

  app.get('/api/admin/build', async () => builder.status());

  app.post('/api/admin/upload', async (req, reply) => {
    const kind = req.query.kind;
    if (kind !== 'image' && kind !== 'video') return reply.code(400).send({ error: 'ชนิดไฟล์ไม่ถูกต้อง' });
    const part = await req.file();
    if (!part) return reply.code(400).send({ error: 'ไม่พบไฟล์' });

    await fsp.mkdir(uploadsDir, { recursive: true });
    await fsp.mkdir(mediaDir, { recursive: true });
    const id = randomUUID();
    const tmp = path.join(uploadsDir, `${id}.upload`);
    const ext = kind === 'image' ? 'webp' : 'mp4';
    const out = path.join(mediaDir, `${id}.${ext}`);
    const fail = async (code, error) => {
      await fsp.rm(tmp, { force: true }); await fsp.rm(out, { force: true });
      return reply.code(code).send({ error });
    };
    try {
      await pipeline(part.file, fs.createWriteStream(tmp));
      const { size } = await fsp.stat(tmp);
      if (part.file.truncated || size > LIMITS[kind]) {
        return fail(413, kind === 'image' ? 'รูปใหญ่เกิน 15 MB' : 'วิดีโอใหญ่เกิน 300 MB');
      }
      if ((await detectKind(tmp)) !== kind) {
        return fail(400, kind === 'image' ? 'ไฟล์ไม่ใช่รูป (รองรับ JPG, PNG, WebP)' : 'ไฟล์ไม่ใช่วิดีโอ (รองรับ MP4, MOV)');
      }
      await processors[kind](tmp, out);
      await fsp.rename(tmp, path.join(uploadsDir, `${id}.orig`));
      return { path: `/media/${id}.${ext}` };
    } catch {
      return fail(500, 'ประมวลผลไฟล์ไม่สำเร็จ กรุณาลองใหม่หรือใช้ไฟล์อื่น');
    }
  });

  return app;
}
```

```js
'use strict';
const $ = (s) => document.querySelector(s);
let content = null, defaults = null, dirty = false, saving = false;

// Tiny DOM helper; everything user-provided goes through textContent/value (no innerHTML).
function h(tag, attrs, ...kids) {
  const el = document.createElement(tag);
  for (const [k, v] of Object.entries(attrs || {})) {
    if (k.startsWith('on')) el.addEventListener(k.slice(2), v); else if (v !== undefined) el.setAttribute(k, v);
  }
  for (const kid of kids.flat()) if (kid != null) el.append(kid.nodeType ? kid : document.createTextNode(kid));
  return el;
}

async function api(method, url, body) {
  const res = await fetch(url, {
    method,
    headers: { 'X-Requested-With': 'kcut-admin', ...(body ? { 'Content-Type': 'application/json' } : {}) },
    body: body ? JSON.stringify(body) : undefined,
    credentials: 'same-origin',
  });
  const data = await res.json().catch(() => ({}));
  if (res.status === 401 && url !== '/api/admin/login') { showLogin(); throw new Error('หมดเวลาการเข้าสู่ระบบ'); }
  if (!res.ok) { const e = new Error(data.error || 'เกิดข้อผิดพลาด'); e.errors = data.errors; throw e; }
  return data;
}

function markDirty() { dirty = true; $('#save').disabled = false; setStatus('มีการแก้ไขที่ยังไม่ได้บันทึก'); }
function setStatus(text, cls) { const s = $('#status'); s.textContent = text; s.className = cls || ''; }

function getPath(path) { return path.reduce((o, k) => o[k], content); }
function setPath(path, v) { getPath(path.slice(0, -1))[path[path.length - 1]] = v; }

function textField(path, label, { multiline = false, number = false } = {}) {
  const val = getPath(path);
  const input = multiline ? h('textarea') : h('input', { type: number ? 'number' : 'text', ...(number ? { min: 1, max: 5 } : {}) });
  input.value = val;
  input.dataset.field = path.map((p) => (typeof p === 'number' ? `[${p}]` : `.${p}`)).join('').replace(/^\./, '');
  input.addEventListener('input', () => { setPath(path, number ? Number(input.value) : input.value); input.classList.remove('invalid'); markDirty(); });
  return h('label', {}, label, input);
}

function mediaField(path, label, kind) {
  const holder = h('div', { class: 'media' });
  const bar = h('div', { class: 'bar', hidden: '' }, h('i'));
  const msg = h('div', { class: 'err' });
  const draw = () => {
    holder.replaceChildren();
    const src = getPath(path);
    holder.append(kind === 'image' ? h('img', { src, alt: '' }) : h('video', { src, muted: '', preload: 'metadata', controls: '' }));
    const file = h('input', { type: 'file', accept: kind === 'image' ? 'image/jpeg,image/png,image/webp' : 'video/mp4,video/quicktime', hidden: '' });
    const btn = h('button', { type: 'button', onclick: () => file.click() }, kind === 'image' ? 'เปลี่ยนรูป' : 'เปลี่ยนวิดีโอ');
    file.addEventListener('change', () => file.files[0] && upload(file.files[0]));
    holder.append(btn, file, bar, msg);
  };
  const upload = (f) => {
    msg.textContent = ''; bar.hidden = false; bar.firstChild.style.width = '0';
    const xhr = new XMLHttpRequest();
    xhr.open('POST', `/api/admin/upload?kind=${kind}`);
    xhr.setRequestHeader('X-Requested-With', 'kcut-admin');
    xhr.upload.onprogress = (e) => { if (e.lengthComputable) bar.firstChild.style.width = `${(e.loaded / e.total) * 100}%`; };
    xhr.onload = () => {
      bar.hidden = true;
      let data = {}; try { data = JSON.parse(xhr.responseText); } catch {}
      if (xhr.status === 401) return showLogin();
      if (xhr.status !== 200) { msg.textContent = data.error || 'อัปโหลดไม่สำเร็จ'; return; }
      setPath(path, data.path); markDirty(); draw();
    };
    xhr.onerror = () => { bar.hidden = true; msg.textContent = 'เชื่อมต่อไม่ได้ กรุณาลองใหม่'; };
    const fd = new FormData(); fd.append('file', f); xhr.send(fd);
  };
  draw();
  return h('div', {}, h('div', { style: 'font-size:.85rem;color:#6b7280;margin-bottom:4px' }, label), holder);
}

function section(title, key, items) {
  const reset = h('button', { type: 'button', onclick: (e) => {
    e.preventDefault();
    if (!confirm(`คืนค่าเดิมของส่วน "${title}" ทั้งหมด?`)) return;
    content[key] = structuredClone(defaults[key]); markDirty(); render();
  } }, 'คืนค่าเดิม');
  return h('details', {}, h('summary', {}, h('h2', {}, title), reset), h('div', { class: 'body' }, items));
}

async function load() {
  const data = await api('GET', '/api/admin/content');
  content = data.content; defaults = data.defaults; dirty = false;
  $('#login').hidden = true; $('#app').hidden = false; $('#bar').hidden = false;
  $('#save').disabled = true; setStatus('');
  render();
}

function showLogin() { $('#app').hidden = true; $('#bar').hidden = true; $('#login').hidden = false; $('#pw').focus(); }

async function pollBuild() {
  setStatus('กำลังเผยแพร่…');
  for (;;) {
    await new Promise((r) => setTimeout(r, 1500));
    const b = await api('GET', '/api/admin/build');
    if (b.state === 'ok') return setStatus('เผยแพร่แล้ว', 'ok');
    if (b.state === 'failed') return setStatus(`เผยแพร่ไม่สำเร็จ: ${b.message}`, 'bad');
  }
}

$('#login').addEventListener('submit', async (e) => {
  e.preventDefault(); $('#loginErr').textContent = '';

  catch (err) { $('#loginErr').textContent = err.message; }
});

$('#logout').addEventListener('click', async () => { await api('POST', '/api/admin/logout').catch(() => {}); showLogin(); });

$('#save').addEventListener('click', async () => {
  if (saving) return; saving = true; $('#save').disabled = true;
  document.querySelectorAll('.invalid').forEach((el) => el.classList.remove('invalid'));
  try {
    await api('PUT', '/api/admin/content', content);
    dirty = false; await pollBuild();
  } catch (err) {
    setStatus(err.message, 'bad'); $('#save').disabled = false;
    for (const e of err.errors || []) {
      const el = document.querySelector(`[data-field="${CSS.escape(e.field)}"]`);
      if (el) { el.classList.add('invalid'); el.title = e.message; }
    }
    if (err.errors?.length) setStatus(`${err.message}: ${err.errors[0].field} ${err.errors[0].message}`, 'bad');
  } finally { saving = false; }
});

window.addEventListener('beforeunload', (e) => { if (dirty) { e.preventDefault(); e.returnValue = ''; } });
```
